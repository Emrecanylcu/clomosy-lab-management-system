void VeritabaniHazirla;
{
  dbYolu = Clomosy.AppFilesPath + 'LabDB_AI_Final.db3';

  if not Clomosy.DBSQLiteConnect(dbYolu, '')
  {
    ShowMessage('Veritabani baglanti hatasi!');
    exit;
  }

  Clomosy.DBSQLiteQuery.Close;
  Clomosy.DBSQLiteQuery.Sql.Text =
    'CREATE TABLE IF NOT EXISTS Musteriler (' +
    'ID INTEGER PRIMARY KEY AUTOINCREMENT, Ad TEXT, KullaniciAdi TEXT, Sifre TEXT, Bakiye INTEGER, LabTipi TEXT)';
  Clomosy.DBSQLiteQuery.OpenOrExecute;

  Clomosy.DBSQLiteQuery.Close;
  Clomosy.DBSQLiteQuery.Sql.Text =
    'CREATE TABLE IF NOT EXISTS Sorunlar (' +
    'ID INTEGER PRIMARY KEY AUTOINCREMENT, MusteriID INTEGER, MusteriAdi TEXT, BilgisayarAdi TEXT, SorunMetni TEXT, Durum TEXT)';
  Clomosy.DBSQLiteQuery.OpenOrExecute;

  Clomosy.DBSQLiteQuery.Close;
  Clomosy.DBSQLiteQuery.Sql.Text =
    'CREATE TABLE IF NOT EXISTS Stok (' +
    'ID INTEGER PRIMARY KEY AUTOINCREMENT, ParcaAdi TEXT, Adet INTEGER, AlisFiyat INTEGER)';
  Clomosy.DBSQLiteQuery.OpenOrExecute;

  Clomosy.DBSQLiteQuery.Close;
  Clomosy.DBSQLiteQuery.Sql.Text =
    'CREATE TABLE IF NOT EXISTS Muhasebe (' +
    'ID INTEGER PRIMARY KEY, ToplamGelir INTEGER, ToplamGider INTEGER, NetKar INTEGER)';
  Clomosy.DBSQLiteQuery.OpenOrExecute;

  Clomosy.DBSQLiteQuery.Close;
  Clomosy.DBSQLiteQuery.Sql.Text =
    'CREATE TABLE IF NOT EXISTS Islemler (' +

}

void MusteriKayitYap;
{
  if EditAd.Text == ''
  {
    ShowMessage('Ad bos olamaz!');
    exit;
  }

  if EditKullanici.Text == ''
  {
    ShowMessage('Kullanici adi bos olamaz!');
    exit;
  }

  if EditSifre.Text == ''
  {
    ShowMessage('Sifre bos olamaz!');
    exit;
  }

  Clomosy.DBSQLiteQuery.Close;
  Clomosy.DBSQLiteQuery.Sql.Text =
    'SELECT * FROM Musteriler WHERE KullaniciAdi=' + QuotedStr(EditKullanici.Text);
  Clomosy.DBSQLiteQuery.OpenOrExecute;

  if Clomosy.DBSQLiteQuery.Found
  {
    Clomosy.DBSQLiteQuery.Close;
    ShowMessage('Bu kullanici adi zaten var!');
    exit;
  }

  Clomosy.DBSQLiteQuery.Close;
  Clomosy.DBSQLiteQuery.Sql.Text =
    'INSERT INTO Musteriler (Ad,KullaniciAdi,Sifre,Bakiye,LabTipi) VALUES (' +
    QuotedStr(EditAd.Text) + ',' +
    QuotedStr(EditKullanici.Text) + ',' +
    QuotedStr(EditSifre.Text) + ',0,' +
    QuotedStr('Standart Lab') + ')';
  Clomosy.DBSQLiteQuery.OpenOrExecute;
  Clomosy.DBSQLiteQuery.Close;

  ShowMessage('Kayit basarili. Giris yapabilirsiniz.');

  MusteriGirisEkrani;
}

void GirisYap;
{
  if GirisTipi == 'AdminLogin'
  {
    if EditKullanici.Text == 'admin'
    {
      if EditSifre.Text == '1234'
        AdminPanel;
      else
        ShowMessage('Sifre hatali!');
    }
    else
      ShowMessage('Kullanici adi hatali!');
  }
  else
  {
    Clomosy.DBSQLiteQuery.Close;
    Clomosy.DBSQLiteQuery.Sql.Text =
      'SELECT * FROM Musteriler WHERE KullaniciAdi=' + QuotedStr(EditKullanici.Text) +
      ' AND Sifre=' + QuotedStr(EditSifre.Text);
    Clomosy.DBSQLiteQuery.OpenOrExecute;

    if Clomosy.DBSQLiteQuery.Found
    {
      Clomosy.DBSQLiteQuery.First;
      AktifMusteriID = Clomosy.DBSQLiteQuery.FieldByName('ID').AsInteger;
      AktifMusteriAdi = Clomosy.DBSQLiteQuery.FieldByName('Ad').AsString;
      Clomosy.DBSQLiteQuery.Close;
      MusteriPanel;
    }
    else
    {
      Clomosy.DBSQLiteQuery.Close;
      ShowMessage('Bilgiler hatali!');
    }
  }
}

void AdminTamirEt;
var
  SorunID, MusteriID, Tutar, Gider, Kar: Integer;
  MusteriAdi, PCAdi, SorunMetni, Parca: String;
{
  Clomosy.DBSQLiteQuery.Close;
  Clomosy.DBSQLiteQuery.Sql.Text =
    'SELECT * FROM Sorunlar WHERE Durum=' + QuotedStr('Bekliyor') + ' ORDER BY ID ASC';
  Clomosy.DBSQLiteQuery.OpenOrExecute;

  if not Clomosy.DBSQLiteQuery.Found
  {
    Clomosy.DBSQLiteQuery.Close;
    ShowMessage('Bekleyen sorun yok.');
    exit;
  }

  Clomosy.DBSQLiteQuery.First;

  SorunID = Clomosy.DBSQLiteQuery.FieldByName('ID').AsInteger;
  MusteriID = Clomosy.DBSQLiteQuery.FieldByName('MusteriID').AsInteger;
  MusteriAdi = Clomosy.DBSQLiteQuery.FieldByName('MusteriAdi').AsString;
  PCAdi = Clomosy.DBSQLiteQuery.FieldByName('BilgisayarAdi').AsString;
  SorunMetni = LowerCase(Clomosy.DBSQLiteQuery.FieldByName('SorunMetni').AsString);

  Clomosy.DBSQLiteQuery.Close;

  Parca = 'Mouse';
  Tutar = 350;
  Gider = 300;

  if Pos('klavye', SorunMetni) > 0
  {
    Parca = 'Klavye';
    Tutar = 515;
    Gider = 450;
  }

  if Pos('monitor', SorunMetni) > 0
  {
    Parca = 'Monitor';
    Tutar = 3170;
    Gider = 2850;
  }

  if Pos('ekran', SorunMetni) > 0
  {
    Parca = 'Monitor';
    Tutar = 3170;
    Gider = 2850;
  }

  if Pos('kasa', SorunMetni) > 0
  {
    Parca = 'Kasa';
    Tutar = 7320;
    Gider = 6600;
  }

  if Pos('kablo', SorunMetni) > 0
  {
    Parca = 'Internet Kablosu';
    Tutar = 262;
    Gider = 225;
  }

  if Pos('internet', SorunMetni) > 0
  {
    Parca = 'Internet Kablosu';
    Tutar = 262;
    Gider = 225;
  }

  Kar = Tutar - Gider;

  Clomosy.DBSQLiteQuery.Close;
  Clomosy.DBSQLiteQuery.Sql.Text =
    'UPDATE Sorunlar SET Durum=' + QuotedStr('Tamamlandi') +
    ' WHERE ID=' + IntToStr(SorunID);
  Clomosy.DBSQLiteQuery.OpenOrExecute;

  Clomosy.DBSQLiteQuery.Close;
  Clomosy.DBSQLiteQuery.Sql.Text =
    'UPDATE Musteriler SET Bakiye = Bakiye + ' + IntToStr(Tutar) +
    ' WHERE ID=' + IntToStr(MusteriID);
  Clomosy.DBSQLiteQuery.OpenOrExecute;

  Clomosy.DBSQLiteQuery.Close;
  Clomosy.DBSQLiteQuery.Sql.Text =
    'UPDATE Muhasebe SET ToplamGelir = ToplamGelir + ' + IntToStr(Tutar) +
    ', ToplamGider = ToplamGider + ' + IntToStr(Gider) +
    ', NetKar = NetKar + ' + IntToStr(Kar) +
    ' WHERE ID=1';
  Clomosy.DBSQLiteQuery.OpenOrExecute;

  Clomosy.DBSQLiteQuery.Close;
  Clomosy.DBSQLiteQuery.Sql.Text =
    'INSERT INTO Islemler (MusteriAdi,BilgisayarAdi,ParcaAdi,ToplamTutar,Kar) VALUES (' +
    QuotedStr(MusteriAdi) + ',' +
    QuotedStr(PCAdi) + ',' +
    QuotedStr(Parca) + ',' +
    IntToStr(Tutar) + ',' +
    IntToStr(Kar) + ')';
  Clomosy.DBSQLiteQuery.OpenOrExecute;

  Clomosy.DBSQLiteQuery.Close;

  ShowMessage('Tamir tamamlandi.');
  AdminPanel;
}
