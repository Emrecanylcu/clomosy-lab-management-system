  SQLKapat;
  Clomosy.DBSQLiteQuery.Sql.Text =
    'UPDATE Talepler SET Durum=' + QuotedStr('Onaylandi') +
    ' WHERE ID=' + IntToStr(TalepID);
  Clomosy.DBSQLiteQuery.OpenOrExecute;

  SQLKapat;
  Clomosy.DBSQLiteQuery.Sql.Text =
    'UPDATE Cihazlar SET Durum=' + QuotedStr('Zimmetli') +
